# 3) استعادة ملف قاعدة مع بقاء -wal قديم من قاعدة أخرى
a = os.path.join(tmp, "a"); os.makedirs(a)
pa = os.path.join(a, "s.sqlite3")
c = sqlite3.connect(pa); c.execute("PRAGMA journal_mode=WAL"); c.execute("PRAGMA wal_autocheckpoint=0")
c.execute("CREATE TABLE t(x)"); c.execute("INSERT INTO t VALUES ('قديم')"); c.commit()
c.execute("PRAGMA wal_checkpoint(TRUNCATE)")
snap = pa + ".snap"; shutil.copy2(pa, snap)                  # «نسخة احتياطية» نظيفة
for i in range(50): c.execute("INSERT INTO t VALUES (?)", (f"جديد{i}",))
c.execute("DELETE FROM t WHERE x='قديم'"); c.commit()        # التغييرات في -wal
# انقطاع: ملفات مفتوحة، ثم «استعادة» ساذجة للملف الرئيسي وحده
wal = pa + "-wal"; walbytes = open(wal, "rb").read()
c.close()  # قد يفرغ WAL عند الإغلاق — نعيد كتابته لمحاكاة الانقطاع
open(wal, "wb").write(walbytes)
shutil.copy2(snap, pa)
try:
    rows = sqlite3.connect(pa).execute("SELECT COUNT(*), SUM(x='قديم') FROM t").fetchone()
    print(f"\n[3] بعد استعادة الملف وحده مع -wal باقٍ: الصفوف={rows[0]}، 'قديم' موجود={bool(rows[1])}"
          f" (المتوقع من النسخة: 1 صف 'قديم')")
except sqlite3.DatabaseError as e:
    print(f"\n[3] بعد استعادة الملف وحده مع -wal باقٍ: قاعدة تالفة — {e}")
