def restart(o, tmp, transport):
    """إعادة تشغيل العملية: كل ما في الذاكرة يضيع (منسق جديد + ذاكرة عدم تكرار فارغة)؛
    القرص والخزينة باقيان."""
    ex = o._conn_executor; ex._http = transport; ex._idempotency = {}
    return runtime.OrchestratorSystem(data_dir=o.data_dir, connector_executor=ex,
                                      org_policy=o.policy.org_policy if hasattr(o.policy, "org_policy") else None,
                                      connector_registry=ex.registry)
class Crash(BaseException): pass
def req(s, title):
    return schemas.new_request(user_id="eval", session_id=s, permissions=["orchestrator:write"],
        utterance=f"افتح مشكلة بعنوان {title} في github.com/acme/webapp")
calls = []
class T(sandbox_env.Transport):
    def __init__(s, crash=False): super().__init__(200); s.crash = crash
    def __call__(s, method, url, **kw):
        calls.append(method.upper())
        if s.crash and method.upper() == "POST": raise Crash("انقطاع العملية بعد وصول الكتابة للطرف البعيد")
        return sandbox_env._Response(200, {"number": 7})

tmp = tempfile.mkdtemp(); o, _ = sandbox_env.build(tmp, transport=T(True))
out = o.submit(req("s1", "عطل")); lid = out["loop_id"]
try: o.decide_approval(lid, out["pending_approval"]["approval_id"], "approved", "owner", False)
except Crash as e: print("انقطاع:", e)
o2 = restart(o, tmp, T(False))
import copy; snap = copy.deepcopy(o2.state.load(lid))  # حالة الانقطاع: running بلا نتيجة

try:
    o2.decide_approval(lid, r0["pending_approval"]["approval_id"], "approved", "owner", False)

except KeyError:
    print("D1 ✗ الاستئناف بعد إعادة التشغيل ينهار: KeyError في ذاكرة الحلقة (داخل العملية فقط)")
